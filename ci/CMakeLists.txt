# CI root for the tools repo (#189): reproduces the monorepo layout the tools CMake expects (workspace/{engine,tools,CMakeLists.txt}) without the private main repo.
# The engine is NOT compiled: the prebuilt package (-DMD_ENGINE_PREBUILT, GitHub Release of the engine repo, commit in ci/ENGINE_SHA) provides monkey_dust::engine; engine/ is
# checked out only for the headers and the two single-file implementations (stb_image_impl.cpp, cgltf_impl.cpp) and the fonts that the tools targets compile in.
cmake_minimum_required(VERSION 3.21)
project(monkey_dust_tools_ci VERSION 0.1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(USE_SDL3 "SDL3 backend" ON)
option(USE_GLM "GLM" OFF)
option(MD_STATIC_DEPS "static deps" OFF)
option(MONKEY_DUST_EDITOR "Editor panels" ON)
set(MD_ENGINE_PREBUILT "" CACHE PATH "Prebuilt engine package")
if(NOT MD_ENGINE_PREBUILT)
    message(FATAL_ERROR "set -DMD_ENGINE_PREBUILT=<unpacked engine package>")
endif()
find_package(SDL3 REQUIRED CONFIG)
find_package(MonkeyDustEngine CONFIG REQUIRED PATHS "${MD_ENGINE_PREBUILT}" NO_DEFAULT_PATH)
file(STRINGS "${CMAKE_SOURCE_DIR}/tools/ci/ENGINE_SHA" _pin LIMIT_COUNT 1)
if(NOT _pin STREQUAL MonkeyDustEngine_SHA)
    message(FATAL_ERROR "package is from ${MonkeyDustEngine_SHA}, tools/ci/ENGINE_SHA pins ${_pin}")
endif()
include(${CMAKE_SOURCE_DIR}/engine/cmake/MdGaiaPch.cmake)
find_library(MD_LUA54_LIB NAMES lua5.4 lua-5.4 REQUIRED)
find_path(MD_LUA54_INC lua5.4/lua.h REQUIRED)
set(LUA_LIBRARIES "${MD_LUA54_LIB}")
set(LUA_INCLUDE_DIR "${MD_LUA54_INC}")
set(MD_FONT_DIR ${CMAKE_SOURCE_DIR}/engine/src/vendor/fonts)
file(GLOB MD_FONT_TTFS ${MD_FONT_DIR}/*.ttf)
enable_testing()   # ctest -R session_synthetic (the synthetic-install test of the editor logic, #185)
add_subdirectory(tools)
